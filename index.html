📊 Conectar Google Sheets
Cada vez que registres una llamada, los datos se envían automáticamente a tu hoja de cálculo con: fecha, cliente, teléfono, monto, resultado, nota y próxima acción.
📋 Pasos (solo una vez):
1
Abrí sheets.google.com → creá una hoja nueva → nombrala "Cobranzas".
2
Menú Extensiones → Apps Script → borrá el código existente → pegá el código de abajo → Guardar.
3
Implementar → Nueva implementación → Aplicación web → Acceso: Cualquier usuario → Implementar → copiá la URL.
4
Pegá la URL acá abajo → Guardar.
Código para Apps Script
function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    
    // Headers si la hoja está vacía
    if (sheet.getLastRow() === 0) {
      var headers = ["Fecha","Tipo contacto","Cliente","Teléfono","Monto ($)","Resultado","Nota","Próxima acción","Fecha próx. contacto","Referencia"];
      var hRow = sheet.getRange(1,1,1,headers.length);
      hRow.setValues([headers]);
      hRow.setFontWeight("bold");
      hRow.setBackground("#0ea5e9");
      hRow.setFontColor("#ffffff");
    }
    
    // Agregar fila con los datos
    var row = [
      data.fecha,
      data.tipoContacto,
      data.cliente,
      data.telefono,
      data.monto,
      data.resultado,
      data.nota,
      data.proximaAccion,
      data.proxContacto,
      data.referencia
    ];
    sheet.appendRow(row);
    sheet.autoResizeColumns(1, 9);
    
    return ContentService
      .createTextOutput(JSON.stringify({ok:true}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService
      .createTextOutput(JSON.stringify({ok:false,error:err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
